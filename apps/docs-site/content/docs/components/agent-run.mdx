## Usage

```tsx
import { AgentRun, AgentRunStep, AgentRunThinking } from "@/components/ui/agent-run";
import { ToolCall } from "@/components/ui/tool-call";
```

```tsx
<AgentRun title="Review project" state="running" defaultOpen>
  <AgentRunStep>Read the brief</AgentRunStep>
  <AgentRunThinking>Comparing the brief with the layout.</AgentRunThinking>
  <ToolCall name="Search files" duration="0.8s" />
  <AgentRunStep state="running">Draft the changes</AgentRunStep>
</AgentRun>
```
