## Usage

```tsx
import { Composer, ComposerInput, ComposerFooter, ComposerAction, ComposerSubmit } from "@/components/ui/composer";
```

```tsx
<Composer status={status} onSubmit={send} onStop={stop}>
  <ComposerInput placeholder="Ask anything" />
  <ComposerFooter>
    <ComposerAction aria-label="Add files">…</ComposerAction>
    <ComposerSubmit className="ms-auto" />
  </ComposerFooter>
</Composer>
```
